```bash
set -euo pipefail
: "${SERVICE_URL:?}" "${RUNNER_HEADERS:?宿主提供权限0600的HTTP头文件}"
KEY="$(python -c 'import uuid; print(uuid.uuid4())')"
printf '%s\n' "$KEY" > /tmp/music-test-idempotency-key
curl --fail-with-body -sS -X POST "$SERVICE_URL/runner/submit" \
  --header @"$RUNNER_HEADERS" \
  -H "Idempotency-Key: $KEY" -H 'Content-Type: application/json' \
  --data-binary @examples/01_generate.submit.json > /tmp/music-submit.json
TASK_ID="$(jq -er '.data.task_id' /tmp/music-submit.json)"
for n in $(seq 1 12); do
  curl --fail-with-body -sS --get "$SERVICE_URL/runner/result" \
    --header @"$RUNNER_HEADERS" \
    --data-urlencode "task_id=$TASK_ID" > /tmp/music-status.json
  test "$(jq -r '.data.finished' /tmp/music-status.json)" = true && break
  sleep 5
done
cat /tmp/music-status.json
# 只有确认ready才下载，不能把pending或JSON错误保存成FLAC。
test "$(jq -r '.data.result.delivery_status' /tmp/music-status.json)" = ready
RESULT_SOURCE_NAME="$(jq -er '.data.result.artifacts[] | select(.media_type=="audio/flac") | .result_source_name' /tmp/music-status.json | head -n 1)"
curl --fail-with-body -sS --get "$SERVICE_URL/runner/result_source" \
  --header @"$RUNNER_HEADERS" \
  --data-urlencode "task_id=$TASK_ID" --data-urlencode "result_source_name=$RESULT_SOURCE_NAME" \
  -o /tmp/music-audio.flac.part
EXPECTED="$(jq -er --arg name "$RESULT_SOURCE_NAME" '.data.result.artifacts[] | select(.result_source_name==$name) | .sha256' /tmp/music-status.json)"
printf '%s  %s\n' "$EXPECTED" /tmp/music-audio.flac.part | sha256sum -c -
```

## v1.2 通用上传与既有资源读取验收补充

仅接口路径POST /runner/upload为新增设计。以下命令在服务实现后执行，本次未执行HTTP或模型测试。上传接口使用file和user_id；普通文本生成可跳过上传。

```bash
# RUNNER_HEADERS是宿主预置、权限0600且不在公开工作区的HTTP头文件。
# 内容由部署端提供，不在命令行展开或回显令牌。
curl --fail-with-body -sS -X POST "$SERVICE_URL/runner/upload" \
  --header @"$RUNNER_HEADERS" \
  -F "user_id=$USER_ID" -F "file=@$INPUT_FILE" > upload.json

# 将upload.json中的asset_id放入相应Task报文再提交。
curl --fail-with-body -sS -X POST "$SERVICE_URL/runner/submit" \
  --header @"$RUNNER_HEADERS" -H 'Content-Type: application/json' \
  --data-binary @submit.json > accepted.json

curl --fail-with-body -sS --get "$SERVICE_URL/runner/result" \
  --header @"$RUNNER_HEADERS" --data-urlencode "task_id=$TASK_ID" > result.json

# RESULT_SOURCE_NAME取result.json真实资源清单中的名称，不猜文件名。
curl --fail-with-body -sS --get "$SERVICE_URL/runner/result_source" \
  --header @"$RUNNER_HEADERS" --data-urlencode "task_id=$TASK_ID" \
  --data-urlencode "result_source_name=$RESULT_SOURCE_NAME" --output artifact.part
# 先校验实际MIME、字节数和SHA256，再将artifact.part改为最终文件名。
```

验收断言：上传不带Task名或environment；同user_id连续上传不删旧文件；user_id与认证身份不符则拒绝；完成前没有可用asset_id；提交不能引用他人asset；result只列已完成文件；result_source_name无法选中元数据或越界路径；断网后重取同一文件，两个GET均不创建Task、不执行发布、不重跑模型。真实文件丢失时明确失败，不能用GET声称修复。
